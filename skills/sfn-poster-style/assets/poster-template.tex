  \ifdim#2in<0pt\PackageError{poster}{Block #1 has a negative x coordinate}{}\fi

    \PackageError{poster}{Block #1 exceeds page height}{Reflow its coordinates.}\fi}

    \PackageError{poster}{Block #1 is too tall}{Shorten text or reflow panel positions.}%
  \fi
  \begin{textblock*}{#4in}(#2in,#3in)\usebox{\layoutbox}\end{textblock*}}
